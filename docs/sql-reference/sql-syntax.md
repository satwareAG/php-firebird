# Firebird SQL Syntax Reference

Complete SQL language reference for Firebird 3.0, 4.0, and 5.0. Organized by
SQL standard categories.

**Convention**: Features introduced after FB 3.0 are tagged with `[FB4]` or
`[FB5]`.

---

## 1. Lexical Structure

### Identifiers

- Regular: `A-Z`, `a-z`, `0-9`, `_`, `$`; max 63 chars `[FB4]` (31 in FB3)
- Must start with a letter
- Case-insensitive (stored uppercase by default)
- Quoted: `"my_field"` - case-sensitive, can contain any character

### Keywords

Reserved keywords cannot be used as identifiers without quoting.

```sql
-- FB5: query all reserved keywords
SELECT RDB$KEYWORD FROM RDB$KEYWORDS ORDER BY 1;  -- [FB5]
```

Source: [keywords @ v5.0.4](https://github.com/FirebirdSQL/firebird/raw/v5.0.4/doc/sql.extensions/README.keywords)

### String Literals

```sql
-- Standard single-quoted
SELECT 'hello world';

-- FB3+ alternate quoting (q'...')
SELECT q'[It's a test]';                    -- "It's a test"
SELECT q'$Path: C:\temp$';                 -- "Path: C:\temp"

-- [FB5] SQL-standard Unicode literals
SELECT U&'caf\00E9';                        -- "cafe"

-- [FB5] SQL-standard binary literals
SELECT X'DEADBEEF';
```

Source: [alternate_string_quoting @ v5.0.4](https://github.com/FirebirdSQL/firebird/raw/v5.0.4/doc/sql.extensions/README.alternate_string_quoting.txt)

### Numeric Literals

```sql
SELECT 42;                  -- INTEGER
SELECT 9223372036854775807; -- BIGINT
SELECT 3.14159;             -- DOUBLE PRECISION
SELECT 0xFF;                -- Hex integer [FB3]
SELECT 0xFFFFFFFFFFFF;      -- Hex for INT128 [FB5]
```

### Comments

```sql
-- Single line comment
/* Multi-line
   comment */
```

## 2. Data Types

### Identity Columns

```sql
CREATE TABLE t (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    -- or:
    id2 INT GENERATED ALWAYS AS IDENTITY
);

-- FB4: restart with value, override system value
ALTER TABLE t ALTER COLUMN id RESTART WITH 100;
INSERT INTO t (id) OVERRIDING SYSTEM VALUE VALUES (999);  -- [FB4]
```

Source: [identity_columns @ v5.0.4](https://github.com/FirebirdSQL/firebird/raw/v5.0.4/doc/sql.extensions/README.identity_columns.txt)

### Domains

```sql
CREATE DOMAIN email_type AS VARCHAR(255)
    DEFAULT ''
    NOT NULL
    CHECK (VALUE CONTAINING '@');

-- Use in table:
CREATE TABLE contacts (id INT, email email_type);
```

## 3. DDL (Data Definition Language)

Source: [ddl @ v5.0.4](https://github.com/FirebirdSQL/firebird/raw/v5.0.4/doc/sql.extensions/README.ddl.txt)

### CREATE TABLE

```sql
CREATE TABLE [IF NOT EXISTS] table_name (
    column_name data_type [DEFAULT value] [NOT NULL]
        [GENERATED ... AS IDENTITY]
        [CHECK (condition)]
        [REFERENCES other_table (col)],

    [CONSTRAINT constraint_name]
        PRIMARY KEY (col, ...)
    |   UNIQUE (col, ...)
    |   FOREIGN KEY (col) REFERENCES ref_table (ref_col)
    |   CHECK (condition)
);

-- SQL SECURITY clause [FB4]
CREATE TABLE secure_data (
    id INT,
    data BLOB
) SQL SECURITY DEFINER;
```

### Indices

```sql
-- Regular index
CREATE [UNIQUE] [ASC[ENDING] | DESC[ENDING]] INDEX idx_name
    ON table_name (col1, col2, ...);

-- Expression index [FB3]
CREATE INDEX idx_email_lower ON contacts (LOWER(email));

-- Partial index [FB5]
CREATE INDEX idx_active ON customers (name)
    WHERE status = 'active';

-- DROP
DROP INDEX idx_name;
```

Sources: [expression_indices @ R3_0_7](https://github.com/FirebirdSQL/firebird/raw/R3_0_7/doc/sql.extensions/README.expression_indices), [partial_indices @ v5.0.4](https://github.com/FirebirdSQL/firebird/raw/v5.0.4/doc/sql.extensions/README.partial_indices)

### Views

```sql
CREATE VIEW view_name AS
    SELECT ... FROM ... WHERE ...;

CREATE OR ALTER VIEW view_name AS
    SELECT ... ;

RECREATE VIEW view_name AS
    SELECT ...;

DROP VIEW view_name;
```

### Procedures / Functions / Packages

```sql
-- Standalone procedure
CREATE PROCEDURE name (params)
    [SQL SECURITY {DEFINER | INVOKER}]  -- [FB4]
    [RETURNS (output_params)]
AS
    [DECLARE ...]
BEGIN
    -- PSQL body
END;

-- Standalone function [FB3]
CREATE FUNCTION name (params)
    RETURNS type
AS
BEGIN
    RETURN value;
END;

-- Package [FB3]
CREATE PACKAGE pkg_name AS
BEGIN
    FUNCTION func_sig(params) RETURNS type;
    PROCEDURE proc_sig(params);
END;

CREATE PACKAGE BODY pkg_name AS
BEGIN
    FUNCTION func_sig(params) RETURNS type AS BEGIN ... END
    PROCEDURE proc_sig(params) AS BEGIN ... END
END;

-- DDL / DROP
CREATE OR ALTER PROCEDURE ...
DROP PROCEDURE name;
DROP FUNCTION name;
DROP PACKAGE pkg_name;
```

Sources: [packages @ R3_0_7](https://github.com/FirebirdSQL/firebird/raw/R3_0_7/doc/sql.extensions/README.packages.txt), [sql_security @ v4.0.7](https://github.com/FirebirdSQL/firebird/raw/v4.0.7/doc/sql.extensions/README.sql_security.txt)

### Triggers

```sql
-- DML triggers
CREATE TRIGGER name FOR table_name
    [ACTIVE | INACTIVE]
    {BEFORE | AFTER} {INSERT | UPDATE | DELETE}
    [POSITION n]
    [SQL SECURITY {DEFINER | INVOKER}]  -- [FB4]
AS
BEGIN
    -- Access NEW.col and OLD.col
END;

-- Multi-event triggers
CREATE TRIGGER name FOR table_name
    BEFORE INSERT OR UPDATE OR DELETE ...

-- Database triggers
CREATE TRIGGER name
    ON {CONNECT | DISCONNECT | TRANSACTION START | TRANSACTION COMMIT}
    [POSITION n]
AS BEGIN ... END;

-- DDL triggers [FB3]
CREATE TRIGGER name ON DDL AS BEGIN ... END;
```

Sources: [universal_triggers @ R3_0_7](https://github.com/FirebirdSQL/firebird/raw/R3_0_7/doc/sql.extensions/README.universal_triggers), [db_triggers @ R3_0_7](https://github.com/FirebirdSQL/firebird/raw/R3_0_7/doc/sql.extensions/README.db_triggers.txt), [ddl_triggers @ R3_0_7](https://github.com/FirebirdSQL/firebird/raw/R3_0_7/doc/sql.extensions/README.ddl_triggers.txt)

### Sequences / Generators

```sql
CREATE SEQUENCE seq_name [START WITH n];
CREATE OR ALTER SEQUENCE seq_name RESTART WITH n;  -- [FB3]
ALTER SEQUENCE seq_name RESTART WITH n;
DROP SEQUENCE seq_name;
RECREATE SEQUENCE seq_name;  -- [FB3]

-- Usage:
SELECT NEXT VALUE FOR seq_name FROM rdb$database;
SELECT GEN_ID(seq_name, 1) FROM rdb$database;  -- legacy syntax
```

Source: [sequence_generators @ R3_0_7](https://github.com/FirebirdSQL/firebird/raw/R3_0_7/doc/sql.extensions/README.sequence_generators)

## 4. DML (Data Manipulation Language)

### SELECT

```sql
SELECT [DISTINCT] columns
FROM tables
[WHERE condition]
[GROUP BY columns [HAVING condition]]
[HAVING condition]                                  -- aggregate filter [FB4]
[ORDER BY columns [NULLS {FIRST | LAST}]]
[ROWS m [TO n]]                                     -- Firebird pagination
[OFFSET m {ROW | ROWS} FETCH {FIRST | NEXT} n {ROW | ROWS} ONLY]  -- [FB3] SQL:2008
[FOR UPDATE [OF columns] [WITH LOCK]]               -- pessimistic locking
[FOR UPDATE [WITH LOCK] [SKIP LOCKED]]              -- [FB5]
[PLAN plan_expression]
[UNION [ALL] | INTERSECT | EXCEPT select_expr]
[WINDOW window_definitions]                         -- [FB4]
```

Sources: [offset_fetch @ R3_0_7](https://github.com/FirebirdSQL/firebird/raw/R3_0_7/doc/sql.extensions/README.offset_fetch.txt), [explicit_locks @ R3_0_7](https://github.com/FirebirdSQL/firebird/raw/R3_0_7/doc/sql.extensions/README.explicit_locks), [skip_locked @ v5.0.4](https://github.com/FirebirdSQL/firebird/raw/v5.0.4/doc/sql.extensions/README.skip_locked.md)

### Window Functions in SELECT

```sql
SELECT
    column1,
    func() OVER (
        [PARTITION BY col, ...]
        [ORDER BY col [ASC|DESC] [NULLS FIRST|LAST], ...]
        [ROWS | RANGE BETWEEN frame_start AND frame_end]  -- [FB4]
    ),
    func() OVER named_window                            -- [FB4]
FROM table
WINDOW named_window AS (window_spec)
;

-- Available functions:
--   ROW_NUMBER(), RANK(), DENSE_RANK()                 [FB3]
--   FIRST_VALUE(), LAST_VALUE(), NTH_VALUE()            [FB3]
--   LAG(), LEAD()                                       [FB3]
--   PERCENT_RANK(), CUME_DIST(), NTILE()                [FB4]
--   SUM(), COUNT(), AVG(), MIN(), MAX() with OVER ()    [FB3]
--   Aggregate with FILTER clause:                       [FB4]
--     SUM(x) FILTER (WHERE condition) OVER (...)
```

Sources: [window_functions @ v5.0.4](https://github.com/FirebirdSQL/firebird/raw/v5.0.4/doc/sql.extensions/README.window_functions.md), [aggregate_filter @ v4.0.7](https://github.com/FirebirdSQL/firebird/raw/v4.0.7/doc/sql.extensions/README.aggregate_filter.md)

### INSERT

```sql
INSERT INTO table [(columns)]
VALUES (values | DEFAULT, ...)          -- DEFAULT keyword [FB4]
[RETURNING cols [INTO :vars]]           -- [FB3] / RETURNING * [FB4]

INSERT INTO table [(columns)]
SELECT ... ;

INSERT INTO table
DEFAULT VALUES;                         -- Insert all defaults

-- Multi-row RETURNING [FB5]
INSERT INTO t SELECT ... FROM ... RETURNING *;
```

### UPDATE

```sql
UPDATE table [AS alias]
SET col = value [, ...]
[WHERE condition]
[PLAN ...]
[ORDER BY ...]
[ROWS n]
[RETURNING cols [INTO :vars]];          -- [FB3] / multi-row RETURNING [FB5]
```

### DELETE

```sql
DELETE FROM table [AS alias]
[WHERE condition]
[PLAN ...]
[ORDER BY ...]
[ROWS n]
[RETURNING cols [INTO :vars]];          -- [FB3] / multi-row RETURNING [FB5]
```

### MERGE

```sql
MERGE INTO target_table [AS t]
USING {source_table | source_query} [AS s]
ON condition
{WHEN MATCHED [AND condition] THEN
    {UPDATE SET ... | DELETE}}
{WHEN NOT MATCHED [BY TARGET] [AND condition] THEN
    INSERT (cols) VALUES (vals)}
{WHEN NOT MATCHED BY SOURCE [AND condition] THEN  -- [FB5]
    {UPDATE SET ... | DELETE}}
[RETURNING cols]                                    -- [FB5]
;
```

Source: [merge @ v5.0.4](https://github.com/FirebirdSQL/firebird/raw/v5.0.4/doc/sql.extensions/README.merge.txt)

### UPDATE OR INSERT

```sql
UPDATE OR INSERT INTO table [(columns)]
VALUES (values)
[MATCHING (columns)]
[RETURNING cols [INTO :vars]];
```

Source: [update_or_insert @ R3_0_7](https://github.com/FirebirdSQL/firebird/raw/R3_0_7/doc/sql.extensions/README.update_or_insert)

### EXECUTE BLOCK

```sql
EXECUTE BLOCK [(input_params)]
[RETURNS (output_params)]
AS
    [DECLARE VARIABLE var type [DEFAULT value];]   -- [FB4] DEFAULT
    [DECLARE FUNCTION sub_func(params) RETURNS type AS BEGIN ... END]  -- [FB3]
BEGIN
    -- PSQL statements
    [SUSPEND;]  -- Return a row when RETURNS is specified
END;
```

Source: [execute_block @ R3_0_7](https://github.com/FirebirdSQL/firebird/raw/R3_0_7/doc/sql.extensions/README.execute_block)

## 5. Transaction Control

Source: [set_transaction @ v5.0.4](https://github.com/FirebirdSQL/firebird/raw/v5.0.4/doc/sql.extensions/README.set_transaction.txt)

```sql
SET TRANSACTION
    [NAME trx_name]                           -- named transactions
    [READ ONLY | READ WRITE]
    [WAIT | NO WAIT]
    [ISOLATION LEVEL {
        READ COMMITTED [{NO | RECORD} VERSION] [READ CONSISTENCY]  -- [FB4]
      | SNAPSHOT [TABLE STABILITY]
    }]
    [AUTO COMMIT]                             -- [FB4]
    [NO AUTO UNDO]
    [LOCK TIMEOUT seconds]
    [RESERVING table_name [FOR [SHARED | EXCLUSIVE] [{READ | WRITE}]]]
    [USING DEFAULT POOL | POOL pool_name];    -- [FB4]

COMMIT [TRANSACTION [trx_name]] [RETAIN [SNAPSHOT]];
ROLLBACK [TRANSACTION [trx_name]] [RETAIN [SNAPSHOT]];

-- Savepoints
SAVEPOINT name;
ROLLBACK [WORK] [TO [SAVEPOINT] name];
RELEASE SAVEPOINT name;
```

Source: [README.read_consistency @ v4.0.7](https://github.com/FirebirdSQL/firebird/raw/v4.0.7/doc/README.read_consistency.md)

## 6. PSQL (Procedural SQL) Syntax

### Control Flow

```sql
IF (condition) THEN
    statement
[ELSE
    statement]

WHILE (condition) DO
    statement;

FOR select_statement INTO :vars DO
    statement;

-- Labels and LEAVE/CONTINUE [FB3]
label_name: LOOP
    IF (condition) THEN LEAVE label_name;
    IF (condition) THEN CONTINUE label_name;
END LOOP;

-- Range-based FOR [FB6 - not in scope for this reference]
```

### Exception Handling

```sql
BEGIN
    -- statements
WHEN {exception_name | GDSCODE code | SQLSTATE state | ANY} DO
BEGIN
    -- error handler
    -- [FB4] Access exception details:
    RDB$GET_CONTEXT('EXCEPTION', 'EXCEPTION_NAME');
    RDB$GET_CONTEXT('EXCEPTION', 'EXCEPTION_MESSAGE');
END
```

Source: [exception_handling @ v5.0.4](https://github.com/FirebirdSQL/firebird/raw/v5.0.4/doc/sql.extensions/README.exception_handling)

### Cursors in PSQL

```sql
DECLARE cursor_name CURSOR FOR (select_statement);

-- Fetch directions:
OPEN cursor_name;
FETCH cursor_name;            -- Forward
FETCH FIRST FROM cursor_name; -- [FB3]
FETCH LAST FROM cursor_name;  -- [FB3]
FETCH PRIOR FROM cursor_name; -- [FB3]
FETCH NEXT FROM cursor_name;  -- [FB3]
FETCH ABSOLUTE n FROM cursor_name; -- [FB3]
FETCH RELATIVE n FROM cursor_name; -- [FB3]
CLOSE cursor_name;
```

Source: [scrollable_cursors @ R3_0_7](https://github.com/FirebirdSQL/firebird/raw/R3_0_7/doc/sql.extensions/README.scrollable_cursors.txt), [cursors @ R3_0_7](https://github.com/FirebirdSQL/firebird/raw/R3_0_7/doc/sql.extensions/README.cursors)

### EXECUTE STATEMENT

```sql
-- Basic
EXECUTE STATEMENT 'SELECT COUNT(*) FROM customers' INTO :cnt;

-- Parameterized
EXECUTE STATEMENT ('SELECT name FROM users WHERE id = ' || :id) INTO :name;

-- [FB4] Rich form (cross-database)
EXECUTE STATEMENT 'SELECT ...'
    ON EXTERNAL 'host:/path/db.fdb'
    [AS USER 'user' PASSWORD 'pass' ROLE 'role']
    [WITH {AUTONOMOUS | COMMON} TRANSACTION]
    [WITH CALLER PRIVILEGES]
    INTO :var;
```

Source: [execute_statement2 @ v4.0.7](https://github.com/FirebirdSQL/firebird/raw/v4.0.7/doc/sql.extensions/README.execute_statement2)

## 10. SET Statements

```sql
SET TRANSACTION ...;                    -- Start transaction
SET AUTODDL ON | OFF;                   -- Auto-DDL in isql
SET PLAN ON | OFF;                      -- Show query plans
SET PLANONLY ON | OFF;                  -- Show plan without executing
SET STATISTICS ON | OFF;                -- Show execution statistics
SET HEADING ON | OFF;                   -- Column headers in isql
SET TIME ZONE 'Europe/Berlin';          -- [FB4]
SET BIND OF type TO type;               -- [FB4]
SET DECFLOAT ROUND TO mode;             -- [FB5]
SET DECFLOAT TRAPS TO trap, ...;        -- [FB5]
SET OPTIMIZE FOR {FIRST_ROWS|ALL_ROWS}; -- [FB5]
SET ROLE role_name;                     -- Activate a role
SET DEFAULT ROLE role_name;             -- [FB4] Set default role for session
```
